  const d = DIGNIDAD[id]; if(!d) return null;
  if(d.dom.includes(signo)) return {t:"domicilio", c:"d-dom"};
  if(d.exa === signo) return {t:"exaltación", c:"d-exa"};

  return {t:"peregrino", c:"d-per"};
}
