function claveActual(){

/* grados que avanza la clave en un año, y años que tarda la vuelta completa */
const porAnio = clave => clave.solar ? 0.98565 : 365.2422/clave.dias;
const vueltaAnios = clave => 360/porAnio(clave);
function puntosMoviles(){
  const l = carta.cuerpos.concat(carta.puntos)
    .map(c => ({id:c.id, nombre:c.nombre, glifo:c.glifo, color:c.color, lon:c.lon}));
  if(carta.ang){
    l.push({id:"asc", nombre:"Ascendente", glifo:PUNTOS.asc.glifo, color:PUNTOS.asc.color, lon:carta.ang.asc});
    l.push({id:"mc",  nombre:"Medio Cielo", glifo:PUNTOS.mc.glifo,  color:PUNTOS.mc.color,  lon:carta.ang.mc});
  }
  return l;
}
/* arco solar real: el Sol avanza un día por cada año de vida */
function arcoSolar(ms){
  const anios = (ms - carta.ms)/(365.2422*86400000);
  return mod360(lonGeo("sol", siglos(julian(carta.ms + anios*86400000))) - carta.cuerpos[0].lon);
}
const arcoDe = (ms, clave) => clave.solar ? arcoSolar(ms) : (ms - carta.ms)/(clave.dias*86400000);
function msDeArco(arco, clave){
  if(!clave.solar) return carta.ms + arco*clave.dias*86400000;
  let t = carta.ms + arco*370.6*86400000;                 // primera aproximación
  for(let i=0;i<10;i++){                                   // y se afina por Newton
    const f = arcoSolar(t) - arco;
    if(Math.abs(f) < 1e-7) break;
    const d = arcoSolar(t + 86400000) - arcoSolar(t);
    if(!isFinite(d) || Math.abs(d) < 1e-10) break;
    t -= (f/d)*86400000;
  }
  return t;
}

/* todos los contactos entre dos arcos, ordenados por fecha */
function contactosAtacir(clave, arcoDesde, arcoHasta){
  const pts = puntosMoviles(), out = [];
  if(arcoHasta - arcoDesde > 1080) arcoHasta = arcoDesde + 1080;   // dos vueltas y media, tope
  for(const P of pts) for(const N of pts){
    if(P.id === N.id) continue;
    for(const asp of ANG_ATACIR){
      const desvios = (asp.a === 0 || asp.a === 180) ? [asp.a] : [asp.a, -asp.a];
      for(const dv of desvios){
        const base = mod360(N.lon + dv - P.lon);
        const k0 = Math.floor((arcoDesde - base)/360), k1 = Math.ceil((arcoHasta - base)/360);
        for(let k = k0; k <= k1; k++){
          const arco = base + 360*k;
          if(arco < arcoDesde || arco > arcoHasta) continue;
          out.push({P, N, asp, arco, ms: msDeArco(arco, clave)});
        }
      }
    }
  }
  return out.sort((a, b) => a.arco - b.arco);
}

/* capa que se dibuja por fuera de la natal */
function capaAtacir(){
  if(!carta) return null;
  const clave = claveActual();
  const fecha = estadoAtacir.fecha == null ? Date.now() : estadoAtacir.fecha;
  const arco = arcoDe(fecha, clave);
  const movidos = puntosMoviles().map(p => Object.assign({}, p, {lonNatal:p.lon, lon:mod360(p.lon + arco)}));
  const cerca = contactosAtacir(clave, arco - ORBE_ATACIR, arco + ORBE_ATACIR);
  return {clave, fecha, arco, movidos, cerca};
}

const fmtDia = new Intl.DateTimeFormat("es-MX", {day:"numeric", month:"long", year:"numeric"});
const fmtCorta = new Intl.DateTimeFormat("es-MX", {day:"2-digit", month:"2-digit", year:"numeric"});
function edadEn(ms){
  const anios = (ms - carta.ms)/(365.2422*86400000);
  if(anios < 0) return "antes de nacer";
  const a = Math.floor(anios), m = Math.round((anios - a)*12);
  return m === 12 ? `${a+1} años` : `${a} a${a === 1 ? "ño" : "ños"}${m ? ` y ${m} ${m === 1 ? "mes" : "meses"}` : ""}`;
}
const arcoTexto = arco => {
  const s = arco < 0 ? "−" : "", v = Math.abs(arco);
  const g = Math.floor(v), m = Math.round((v - g)*60);
  return `${s}${m === 60 ? g+1 : g}° ${String(m === 60 ? 0 : m).padStart(2,"0")}′`;
};
