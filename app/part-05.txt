80053 + 180); }
function oblicuidad(T){ return 23.4392911 - 0.0130042*T - 1.64e-7*T*T + 5.04e-7*T*T*T; }
function gmst(jd){
  const T = (jd-2451545)/36525;
  return mod360(280.46061837 + 360.98564736629*(jd-2451545) + 0.000387933*T*T - T*T*T/38710000);
}

/* --- ángulos y casas --- */
function angulos(jd, lat, lon){
  const T = siglos(jd), eps = oblicuidad(T)*RAD, phi = lat*RAD;
  const ramcDeg = mod360(gmst(jd) + lon), ramc = ramcDeg*RAD;
  const mc = mod360(Math.atan2(Math.sin(ramc), Math.cos(ramc)*Math.cos(eps))*DEG);
  let asc = mod360(Math.atan2(Math.cos(ramc), -(Math.sin(ramc)*Math.cos(eps) + Math.tan(phi)*Math.sin(eps)))*DEG);
  if(mod360(asc - mc) > 180) asc = mod360(asc + 180);
  return {mc, asc, ramc:ramcDeg, eps:eps*DEG, lat, lon};
}
function placidio(ramcDeg, latDeg, epsDeg){
  const eps = epsDeg*RAD, phi = latDeg*RAD;
  const lonDeRA = a => mod360(Math.atan2(Math.sin(a*RAD), Math.cos(a*RAD)*Math.cos(eps))*DEG);
  function cuspide(inicio, f, nocturna){
    let alpha = ramcDeg + inicio;
    for(let i=0;i<30;i++){
      const lam = lonDeRA(alpha)*RAD;
      const dec = Math.asin(Math.sin(eps)*Math.sin(lam));
      const x = -Math.tan(phi)*Math.tan(dec);
      if(Math.abs(x) >= 1) return null;                 // círculo polar: sin solución
      const D = Math.acos(x)*DEG;                       // semiarco diurno
      const nuevo = nocturna ? ramcDeg + 180 - f*(180-D) : ramcDeg + f*D;
      if(Math.abs(nuevo-alpha) < 1e-9){ alpha = nuevo; break; }
      alpha = nuevo;
    }
    return lonDeRA(alpha);
  }
  const c11 = cuspide(30,1/3,false), c12 = cuspide(60,2/3,false);
  const c2 = cuspide(120,2/3,true),  c3 = cuspide(150,1/3,true);
  return [c11,c12,c2,c3].some(v => v === null) ? null : {c11,c12,c2,c3};
}
function cuspides(sistema, ang){
  const c = new Array(13);
  const {asc, mc} = ang;
  if(sistema === "signos"){
    const base = Math.floor(asc/30)*30;
    for(let i=1;i<=12;i++) c[i] = mod360(base + (i-1)*30);
    return {c, sistema:"signos"};
  }
  if(sistema === "igual"){
    for(let i=1;i<=12;i++) c[i] = mod360(asc + (i-1)*30);
    return {c, sistema:"igual"};
  }
  if(sistema === "placidio"){
    const p = placidio(ang.ramc, ang.lat, ang.eps);
    if(p){
      c[1]=asc; c[10]=mc; c[11]=p.c11; c[12]=p.c12; c[2]=p.c2; c[3]=p.c3;
      c[4]=mod360(mc+180); c[7]=mod360(asc+180);
      c[5]=mod360(p.c11+180); c[6]=mod360(p.c12+180);
      c[8]=mod360(p.c2+180); c[9]=mod360(p.c3+180);
      return {c, sistema:"placidio"};
    }
    return Object.assign(cuspides("signos", ang), {aviso:"Placidio no tiene solución más allá del círculo polar; se usaron signos enteros."});
  }
  // porfirio: cada cuadrante partido en tres
  c[1]=asc; c[10]=mc; c[4]=mod360(mc+180); c[7]=mod360(asc+180);
  const q1 = mod360(asc - mc)/3, q2 = mod360(mod360(mc+180) - asc)/3;
  c[11]=mod360(mc+q1); c[12]=mod360(mc+2*q1);
  c[2]=mod360(asc+q2); c[3]=mod360(asc+2*q2);
  c[5]=mod360(c[11]+180); c[6]=mod360(c[12]+180);
  c[8]=mod360(c[2]+180);  c[9]=mod360(c[3]+180);
  return {c, sistema:"porfirio"};
}
function casaDe(lon, c){
  if(!c) return null;
  for(let i=1;i<=12;i++){
    const a = c[i], b = c[i===12?1:i+1];
    if(mod360(lon-a) < mod360(b-a)) return i;
  }
  return 1;
}

function calculaAspectos(lista, factor){
  const res = [];
  for(let i=0;i<lista.length;i++) for(let j=i+1;j<lista.length;j++){
    const A = lista[i], B = lista[j];
    let d = Math.abs(mod360(A.lon - B.lon)); if(d > 180) d = 360 - d;
    for(const asp of ASPECTOS){
      const luminaria = ["sol","luna"].includes(A.id) || ["sol","luna"].includes(B.id);
      const orbe = (asp.orbe + (luminaria && asp.mayor ? 2 : 0)) * factor;
      const dif = Math.abs(d - asp.a);
      if(dif <= orbe){ res.push({A, B, asp, dif}); break; }
    }
  }
  return res.sort((x,y) => x.dif - y.dif);
}

};

/* --- zona horaria: el navegador trae la base histórica (incluye horarios de verano) --- */
function utcMs(y,mo,d,h,mi){
  const t = Date.UTC(y, mo-1, d, h, mi, 0);
  if(y >= 0 && y < 100){ const f = new Date(t); f.setUTCFullYear(y); return f.getTime(); }
  return t;
}
function desfaseZona(tz, ms){
  const dtf = new Intl.DateTimeFormat("en-US",{timeZone:tz, hourCycle:"h23",
    year:"numeric", month:"2-digit", day:"2-digit", hour:"2-digit", minute:"2-digit", second:"2-digit"});
  const p = Object.fromEntries(dtf.formatToParts(new Date(ms)).map(o => [o.type, o.value]));
  return utcMs(+p.year, +p.month, +p.day, +p.hour, +p.minute) + (+p.second)*1000 - ms;
}
function localAUTC(y,mo,d,h,mi,tz){
  const base = utcMs(y,mo,d,h,mi);
  let t = base;
  for(let i=0;i<4;i++) t = base - desfaseZona(tz, t);
  return t;
}
function textoDesfase(tz, ms){
  const min = Math.round(desfaseZona(tz, ms)/60000);
  const s = min < 0 ? "−" : "+", a = Math.abs(min);
  return `UTC${s}${String(Math.floor(a/60)).padStart(2,"0")}:${String(a%60).padStart(2,"0")}`;
}

            lon, retro: (((lon - antes + 540) % 360) - 180) < 0, tipo:"astro"};
  });
  const nn = nodoNorte(T);
  const puntos = [
    {id:"nodoN", nombre:PUNTOS.nodoN.nombre, glifo:PUNTOS.nodoN.glifo, color:PUNTOS.nodoN.color, lon:nn, retro:true, tipo:"punto"},
    {id:"nodoS", nombre:PUNTOS.nodoS.nombre, glifo:PUNTOS.nodoS.glifo, color:PUNTOS.nodoS.color, lon:mod360(nn+180), retro:true, tipo:"punto"},
    {id:"lilith",nombre:PUNTOS.lilith.nombre,glifo:PUNTOS.lilith.glifo,color:PUNTOS.lilith.color,lon:lilithMedia(T), retro:false, tipo:"punto"}
  ];

  let ang = null, cusp = null, aviso = null, fortuna = null, diurna = null;
  if(d.horaConocida){
    ang = angulos(jd, d.lat, d.lon);
    const r = cuspides(d.sistema, ang);
    cusp = r.c; aviso = r.aviso || null; d.sistemaReal = r.sistema;
    const sol = cuerpos[0], luna = cuerpos[1];
    const casaSol = casaDe(sol.lon, cusp);
    diurna = casaSol >= 7 && casaSol <= 12;
    const lonFort = diurna ? mod360(ang.asc + luna.lon - sol.lon) : mod360(ang.asc + sol.lon - luna.lon);
    fortuna = {id:"fortuna", nombre:PUNTOS.fortuna.nombre, glifo:PUNTOS.fortuna.glifo,
               color:PUNTOS.fortuna.color, lon:lonFort, retro:false, tipo:"punto"};
    puntos.push(fortuna);
  }

  const paraAspectos = cuerpos.concat(puntos.filter(p => p.id === "nodoN"));
  if(ang){
    paraAspectos.push({id:"asc", nombre:"Ascendente", glifo:PUNTOS.asc.glifo, color:PUNTOS.asc.color, lon:ang.asc, tipo:"angulo"});
    paraAspectos.push({id:"mc",  nombre:"Medio Cielo", glifo:PUNTOS.mc.glifo,  color:PUNTOS.mc.color,  lon:ang.mc,  tipo:"angulo"});
  }
  const aspectos = calculaAspectos(paraAspectos, d.factorOrbe);

  carta = {datos:d, ms, jd, cuerpos, puntos, ang, cusp, aspectos, aviso, diurna};
  return carta;
}
