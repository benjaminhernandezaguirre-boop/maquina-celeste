0,W,H);
  if(nebulosa) ctx.drawImage(nebulosa,0,0);
  const t = ms/1000;
  for(const s of estrellas){
    const a = s.a * (0.55 + 0.45*Math.sin(t*s.f + s.d));
    ctx.globalAlpha = a; ctx.fillStyle = s.t;
    ctx.beginPath(); ctx.arc(s.x, s.y, s.r, 0, 7); ctx.fill();
  }
  ctx.globalAlpha = 1;
}

/* ---- esfera con brillo ---- */
function esfera(x,y,r,color,brillo=1){
  const g = ctx.createRadialGradient(x-r*.35, y-r*.4, r*.08, x, y, r*(2.9*brillo));
  g.addColorStop(0, color); g.addColorStop(.22, color+"66"); g.addColorStop(1, "rgba(0,0,0,0)");
  ctx.fillStyle = g; ctx.beginPath(); ctx.arc(x,y,r*3*brillo,0,7); ctx.fill();
  const b = ctx.createRadialGradient(x-r*.4, y-r*.45, r*.1, x, y, r);
  b.addColorStop(0,"#FFFFFF"); b.addColorStop(.25,color); b.addColorStop(1, mezcla(color,"#000000",.55));
  ctx.fillStyle = b; ctx.beginPath(); ctx.arc(x,y,r,0,7); ctx.fill();
}
function mezcla(hex, hex2, k){
  const p = h => [parseInt(h.slice(1,3),16),parseInt(h.slice(3,5),16),parseInt(h.slice(5,7),16)];
  const a = p(hex), b = p(hex2);
  return "#"+a.map((v,i)=>Math.round(v+(b[i]-v)*k).toString(16).padStart(2,"0")).join("");
}
function etiqueta(x,y,txt,color,activo){
  ctx.font = `${activo?500:400} 11px "IBM Plex Mono", monospace`;
  ctx.textAlign = "center"; ctx.textBaseline = "top";
  ctx.fillStyle = activo ? "#EDE4D2" : color+"CC";
  ctx.shadowColor = "rgba(0,0,0,.9)"; ctx.shadowBlur = 6;
  ctx.fillText(txt, x, y);
  ctx.shadowBlur = 0;
}

/* ---- vista de órbitas (heliocéntrica, escala comprimida y perspectiva) ---- */
const compr = r => Math.log(1 + r/0.25);            // escala logarítmica: los interiores respiran
const COMP_MAX = compr(41);
const VISTA_ORBITAL = {inclinacion:.52, giro:-.18};
function proyOrbital(x,y,z,k){
  const r = Math.hypot(x,y,z);
  if(r === 0) return [0,0,0,1];
  const rr = compr(r)*k;
  let X=x/r*rr, Y=y/r*rr, Z=z/r*rr;
  // Giramos la eclíptica y luego la inclinamos: se perciben los planos orbitales reales.
  const cg=Math.cos(VISTA_ORBITAL.giro), sg=Math.sin(VISTA_ORBITAL.giro);
  const xg=X*cg-Y*sg, yg=X*sg+Y*cg;
  const ci=Math.cos(VISTA_ORBITAL.inclinacion), si=Math.sin(VISTA_ORBITAL.inclinacion);
  const yp=yg*ci-Z*si, zp=yg*si+Z*ci;
  const persp = Math.max(.82,Math.min(1.18,1-zp/(Math.max(W,H)*2.7)));
  return [xg*persp,yp*persp,zp,persp];
}
function centro(){ return [W/2 - desvio(), H/2]; }
function desvio(){
  const panel = vista === "natal"
    ? document.getElementById("cartaPanel").classList.contains("abierto")
    : ficha.classList.contains("abierta");
  return (panel && W > 900) ? Math.min(200, W*0.145) : 0;
}
function dibujaOrbitas(T, ms){
  const [cx, cy] = centro();
  const k = Math.min(W,H)/2 * 0.91 * zoom / COMP_MAX;

  // Halo solar y disco eclíptico tenue: ayuda a leer profundidad sin inventar posiciones.
  const halo = ctx.createRadialGradient(cx,cy,0,cx,cy,Math.min(W,H)*.42);
  halo.addColorStop(0,"rgba(255,207,110,.095)"); halo.addColorStop(.24,"rgba(227,190,117,.025)"); halo.addColorStop(1,"rgba(0,0,0,0)");
  ctx.fillStyle=halo; ctx.fillRect(0,0,W,H);

  // Órbitas: usamos x/y/z reales de los elementos orbitales y las proyectamos con perspectiva.
  ctx.lineCap="round";
  for(const a of ASTROS){
    if(!a.orbita) continue;
    const sel = seleccion === a.id || sobre === a.id;
    const per = ELEM[a.orbita][9];
    const paso = 36525/Math.abs(per)*360/144;
    let prev=null;
    for(let i=0;i<=144;i++){
      const p = helio(a.orbita, T + (i*paso)/36525);
      const pr = proyOrbital(p.x,p.y,p.z,k);
      if(prev){
        const depth=(prev[2]+pr[2])*.5;
        const normal=.5+.5*Math.tanh(depth/(Math.min(W,H)*.18));
        const alpha = sel ? (.28+.30*normal) : (.045+.085*normal);
        ctx.strokeStyle = sel ? a.color+Math.round(alpha*255).toString(16).padStart(2,"0") : `rgba(237,228,210,${alpha})`;
        ctx.lineWidth = sel ? 1.35 : .85;
        ctx.beginPath(); ctx.moveTo(cx+prev[0],cy-prev[1]); ctx.lineTo(cx+pr[0],cy-pr[1]); ctx.stroke();
      }
      prev=pr;
    }
  }

  // Sol: pulso mínimo, como variación de brillo y no como cambio de tamaño físico.
  const pul = 1 + 0.018*Math.sin(ms/1500);
  esfera(cx, cy, porId.sol.radio*pul, porId.sol.color, 1.75);
  ctx.strokeStyle="#E3BE7544"; ctx.lineWidth=1;
  ctx.beginPath(); ctx.arc(cx,cy,porId.sol.radio*2.35,0,7); ctx.stroke();
  blancos.push({id:"sol", x:cx, y:cy, r:26});
  etiqueta(cx, cy+31, "SOL", porId.sol.color, seleccion==="sol");

  // Calculamos todos los planetas primero y los ordenamos por profundidad.
  const visibles=[];
  for(const a of ASTROS){
    if(!a.orbita) continue;
    const p = helio(a.orbita, T);
    const pr = proyOrbital(p.x,p.y,p.z,k);
    visibles.push({a,p,pr});
  }
  visibles.sort((A,B)=>A.pr[2]-B.pr[2]);

  for(const {a,pr} of visibles){
    const [px,py,z,ps] = pr;
    const x = cx+px, y = cy-py;
    const act = seleccion === a.id || sobre === a.id;
    const profundidad = Math.max(.82,Math.min(1.16,ps));
    const rr = a.radio*profundidad*(act?1.12:1);

    // Pequeña sombra de contacto y halo: mejora la lectura de cada cuerpo contra la nebulosa.
    ctx.fillStyle="rgba(0,0,0,.22)"; ctx.beginPath(); ctx.ellipse(x+2,y+rr*.7,rr*.9,rr*.32,0,0,7); ctx.fill();
    esfera(x, y, rr, a.color, act?1.30:.92);

    if(a.anillo){
      ctx.save(); ctx.translate(x,y); ctx.rotate(-.35); ctx.scale(1,.31);
      ctx.strokeStyle = a.color+"BB"; ctx.lineWidth = 2.2;
      ctx.beginPath(); ctx.arc(0,0,rr*1.95,0,7); ctx.stroke();
      ctx.strokeStyle = a.color+"52"; ctx.lineWidth = 1.1;
      ctx.beginPath(); ctx.arc(0,0,rr*2.45,0,7); ctx.stroke();
      ctx.restore();
    }

    blancos.push({id:a.id, x, y, r:Math.max(16,rr+9)});
    etiqueta(x, y+rr+8, a.nombre.toUpperCase(), a.color, act);

    if(a.id === "tierra"){
      const ll = lunaLon(T)*RAD;
      const d = 22*Math.max(.75,zoom)*profundidad;
      const lx = x + Math.cos(ll)*d, ly = y - Math.sin(ll)*d*.82;
      ctx.strokeStyle = "#FFFFFF18"; ctx.lineWidth = 1;
      ctx.beginPath(); ctx.ellipse(x,y,d,d*.82,0,0,7); ctx.stroke();
      esfera(lx, ly, 4*profundidad, porId.luna.color, .82);
      blancos.push({id:"luna", x:lx, y:ly, r:13});
    }
  }
}

/* ---- vista de rueda zodiacal (geocéntrica) ---- */
function dibujaRueda(T){
  const [cx, cy] = centro();
  const R = Math.min(W - 2*desvio(), H)/2 * 0.9 * zoom;
  const rExt = R, rInt = R*0.80, rPl = R*0.66, rCen = R*0.30;
  const aPant = lon => (180 - lon) * RAD;           // 0° Aries a la izquierda, sentido antihorario

  // sectores de los doce signos
  for(let i=0;i<12;i++){
    const a0 = aPant(i*30), a1 = aPant(i*30+30);
    const col = COLOR_EL[SIGNOS[i].el];
    ctx.beginPath();
    ctx.arc(cx,cy,rExt,a1,a0); ctx.arc(cx,cy,rInt,a0,a1,true); ctx.closePath();
    ctx.fillStyle = col+"1F"; ctx.fill();
    ctx.strokeStyle = "#2A2E4E"; ctx.lineWidth = 1; ctx.stroke();

    const am = aPant(i*30+15), rm = (rExt+rInt)/2;
    const gx = cx+Math.cos(am)*rm, gy = cy+Math.sin(am)*rm;
    ctx.fillStyle = col; ctx.textAlign = "center"; ctx.textBaseline = "middle";
    ctx.font = `${Math.round(R*0.075)}px Spectral, serif`;
    ctx.fillText(SIGNOS[i].g, gx, gy - R*0.026);
    ctx.fillStyle = "#8E8CAB"; ctx.font = `${Math.max(8,Math.round(R*0.028))}px "IBM Plex Mono", monospace`;
    ctx.fillText(SIGNOS[i].n.toUpperCase(), gx, gy + R*0.045);
  }
  // grados cada 5°
  ctx.strokeStyle = "#FFFFFF22";
  for(let d=0; d<360; d+=5){
    const a = aPant(d), largo = d%30===0 ? R*0.055 : R*0.022;
    ctx.beginPath();
    ctx.moveTo(cx+Math.cos(a)*rInt, cy+Math.sin(a)*rInt);
    ctx.lineTo(cx+Math.cos(a)*(rInt-largo), cy+Math.sin(a)*(rInt-largo));
    ctx.stroke();
  }
  // Tierra al centro
  esfera(cx, cy, Math.max(8,R*0.045), porId.tierra.color, 1.1);
  blancos.push({id:"tierra", x:cx, y:cy, r:R*0.07});
  ctx.fillStyle="#6C6A88"; ctx.textAlign="center"; ctx.textBaseline="top";
  ctx.font = `9px "IBM Plex Mono", monospace`;
  ctx.fillText("TIERRA", cx, cy+R*0.06);

  // astros sobre la rueda, separando los que se apiñan
  const usados = [];
  for(const id of ORDEN_EFEM){
    const a = porId[id], lon = lonGeo(id, T), z = posZod(lon);
    let radio = rPl;
    while(usados.some(u => Math.abs(((u.lon-lon+540)%360)-180) < 9 && Math.abs(u.r-radio) < R*0.05)) radio -= R*0.062;
    usados.push({lon, r:radio});
    const ang = aPant(lon);
    const x = cx+Math.cos(ang)*radio, y = cy+Math.sin(ang)*radio;

    ctx.strokeStyle = a.color+"3A"; ctx.lineWidth = 1;
    ctx.beginPath();
    ctx.moveTo(cx+Math.cos(ang)*(rInt-R*0.01), cy+Math.sin(ang)*(rInt-R*0.01));
    ctx.lineTo(x,y); ctx.stroke();

    const act = seleccion===id || sobre===id;
    ctx.fillStyle = act ? "#EDE4D2" : a.color;
    ctx.shadowColor = a.color; ctx.shadowBlur = act ? 20 : 10;
    ctx.textAlign="center"; ctx.textBaseline="middle";
    pintaGlifo(ctx, a.glifo, x, y, Math.round(R*0.062));
    ctx.shadowBlur = 0;
    ctx.fillStyle = act ? "#EDE4D2" : "#9A98B6";
    ctx.font = `${Math.max(8,Math.round(R*0.027))}px "IBM Plex Mono", monospace`;
    ctx.fillText(z.corto, x, y + R*0.055);
    blancos.push({id, x, y, r:Math.max(17,R*0.05)});
  }
}

function enPunto(ev){
  const r = lienzo.getBoundingClientRect();
  const x = ev.clientX - r.left, y = ev.clientY - r.top;
  let mejor = null, dm = 1e9;
  for(const b of blancos){
    const d = Math.hypot(b.x-x, b.y-y);
    if(d < b.r && d < dm){ dm = d; mejor = b; }
  }
  return mejor;
}

  const cielo = document.getElementById("portadaCielo");
  if(cielo && !portada.classList.contains("oculta")){
    const zs = posZod(lonGeo("sol", T)), zl = posZod(lonGeo("luna", T));
    cielo.textContent = `En este momento · Sol ${zs.texto} ${zs.signo.n} · Luna ${zl.texto} ${zl.signo.n}`;
  }
  for(const id of ORDEN_EFEM){
    const lon = lonGeo(id, T), z = posZod(lon);
    const antes = lonGeo(id, T - 1/36525);
    const retro = ((lon - antes + 540) % 360) - 180 < 0;
    const el = document.getElementById("ef-"+id);
    if(el) el.innerHTML = `${z.corto} ${z.signo.g}${retro?' <span class="r">℞</span>':""}`;
  }
}

  efem.querySelectorAll("button").forEach(b => b.classList.toggle("sel", b.dataset.id === seleccion));
}
